5. File Permissions

ls -l file.txt            # Check permissions
chmod 777 file.txt        # Give all permissions to everyone
chmod 755 file.txt        # Owner full, others read and execute
chmod 644 file.txt        # Owner read/write, others read
chmod +x script.sh        # Add execute permission
chmod -x script.sh        # Remove execute permission
chmod u+x script.sh      # Add execute for owner
chmod g+w file.txt        # Add group write permission
chmod o-r file.txt        # Remove others read permission
chown user file.txt       # Change file owner
chgrp group file.txt      # Change group ownership
umask                     # Display default permission mask

sh script.sh                   # Execute using sh
chmod +x script.sh             # Make executable

echo $name                     # Print variable

sleep 5                        # Wait 5 seconds
date                           # Display date

Example shell script:


#!/bin/bash

echo "Enter your name:"
read name

echo "Hello $name"

echo "Current date:"
date
